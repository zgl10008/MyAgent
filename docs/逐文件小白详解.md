## app/main.py — 程序的「开关」

**作用**：整个网站的入口。按下这个开关，网站就启动了。

### 逐行白话解释

```python
from fastapi import FastAPI
# FastAPI 是一个 Python 库，帮你快速搭建网站（类似 Flask，但更快，自带 API 文档）

from app.api.routes import chat, document, health, sessions
# 导入 4 个「路由模块」，它们是网站的不同页面/功能

from app.config import get_settings
# 导入获取配置的函数

from app.infrastructure.database.session import configure_session, init_engine
# 导入数据库相关函数

@asynccontextmanager
async def lifespan(app: FastAPI):
    """
    这个函数定义了「应用的一生」：
    - lifespan 之前 = 出生时做什么
    - yield 之前 = 启动时做什么
    - yield 之后 = 关闭时做什么
    用 @asynccontextmanager 装饰器，FastAPI 会自动在启动/关闭时调用
    """
    settings = get_settings()
    # 读取 .env 文件中的配置

    logger.info("启动 {} ({})", settings.app_name, settings.app_env)
    # 打印启动日志

    engine = init_engine(settings.database_url)
    # 创建数据库连接引擎（但还没连上）

    configure_session(engine)
    # 把引擎存为全局变量，后面所有代码都能用

    # 自动建表（只在开发环境，生产环境应该用 Alembic 迁移工具）
    try:
        from app.infrastructure.database.models import Base
        async with engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
        # Base.metadata.create_all 会读取上面定义的表模型，在 PG 里创建对应的表
    except Exception as exc:
        logger.warning("数据库建表失败，可能 PG 未启动: {}", exc)
        # PG 没启动也不崩溃，因为项目有内存降级方案

    yield
    # ----- 上面是启动代码，下面是关闭代码 -----

    await engine.dispose()
    # 关闭数据库连接，释放资源

def create_app() -> FastAPI:
    """工厂函数：创建并配置 FastAPI 应用"""
    application = FastAPI(
        title=settings.app_name,   # 网站标题
        debug=settings.debug,      # 调试模式
        lifespan=lifespan,         # 绑定生命周期
    )

    # CORS 配置：允许前端跨域访问
    # 简单说：前端在 localhost:5173，后端在 localhost:8000
    # 浏览器默认不允许不同端口的 AJAX 请求，CORS 就是告诉浏览器「允许」
    application.add_middleware(
        CORSMiddleware,
        allow_origins=["http://localhost:5173", "http://127.0.0.1:5173"],
        allow_credentials=True,
        allow_methods=["*"],   # 允许所有 HTTP 方法
        allow_headers=["*"],   # 允许所有 HTTP 头
    )

    # 注册路由：把 4 个功能模块挂到 /api/v1 下
    # 比如 health 路由的 /health 最终变成 /api/v1/health
    application.include_router(health.router, prefix=settings.api_prefix)
    application.include_router(chat.router, prefix=settings.api_prefix)
    application.include_router(document.router, prefix=settings.api_prefix)
    application.include_router(sessions.router, prefix=settings.api_prefix)
    return application

app = create_app()
# 创建全局 app 实例，uvicorn 就启动这个
```

### 启动命令解释

```bash
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
# uvicorn    : ASGI 服务器（类似 nginx 之于网页）
# app.main   : 文件路径 app/main.py
# :app       : 文件里的 app 变量（就是上面 create_app() 的返回值）
# --reload   : 代码改了自动重启
# --host 0.0.0.0 : 允许任何 IP 访问（局域网也能连）
# --port 8000    : 监听 8000 端口
```

## app/config.py — 程序的「遥控器」

**作用**：集中管理所有配置。改了 .env 文件，程序行为就变了。

### 逐行白话解释

```python
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):
    """
    这个类继承 BaseSettings，会自动从 .env 文件读取配置。
    比如 .env 里写了 OPENAI_API_KEY=sk-xxx，
    那 Settings().openai_api_key 的值就是 "sk-xxx"。
    """

    model_config = SettingsConfigDict(
        env_file=".env",              # 从哪个文件读
        env_file_encoding="utf-8",    # 文件编码
        extra="ignore",               # .env 里多写的字段不报错
    )

    # 下面每一个属性 = 一个配置项
    # Field(default=xxx, description=...) 定义了默认值和说明
    app_name: str = Field(default="enterprise-ai-agent")
    app_env: str = Field(default="development")     # development | production
    debug: bool = Field(default=False)
    api_prefix: str = Field(default="/api/v1")      # 所有 API 的前缀
    host: str = Field(default="0.0.0.0")
    port: int = Field(default=8000)

    # === LLM 配置（最重要！） ===
    openai_api_key: str = Field(default="", description="OpenAI API Key")
    # ↑ .env 里必须填，否则所有 AI 功能不可用
    openai_api_base: str = Field(default="https://api.openai.com/v1")
    # ↑ 如果用 DeepSeek，改成 https://api.deepseek.com
    openai_model: str = Field(default="gpt-4o-mini")
    # ↑ 用哪个模型（gpt-4o-mini / deepseek-v4-flash / qwen-turbo）

    # === 数据库 ===
    database_url: str = Field(default="postgresql+asyncpg://postgres:postgres@localhost:5432/agent_db")
    # ↑ PG 的连接地址：格式是 协议://用户名:密码@主机:端口/数据库名

    # === Redis ===
    redis_url: str = Field(default="redis://localhost:6379/0")

    # === 向量库 ===
    milvus_db_path: str = Field(default="milvus.db")
    # ↑ Milvus Lite 把向量数据存在本地文件里，不需要 Docker

    # === 搜索 ===
    serpapi_api_key: str = Field(default="", description="SerpAPI 搜索 API Key")
    # ↑ Agent 搜索工具需要，免费注册 https://serpapi.com

    # === HuggingFace 镜像（国内用户加速下载模型） ===
    hf_endpoint: str = Field(default="https://hf-mirror.com")

    log_level: str = Field(default="INFO")

@lru_cache
def get_settings() -> Settings:
    """
    @lru_cache 是 Python 内置的缓存装饰器。
    第一次调用 → 创建 Settings 对象 → 缓存
    第二次调用 → 直接返回缓存的对象，不会重新读 .env
    这保证全局只有一个 Settings 实例（单例模式）。
    """
    return Settings()
```

## app/models/schemas.py — 数据的「身份证」

**作用**：定义所有输入数据和输出数据的「长相」。FastAPI 靠这些模型自动校验请求、生成 API 文档。

### 先理解：Pydantic 是什么

```python
# 普通 Python 类：
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

# Pydantic 类（自动校验类型）：
from pydantic import BaseModel, Field

class Person(BaseModel):
    name: str = Field(description="姓名")
    age: int = Field(ge=0, le=150, description="年龄")

# 用法区别：
# 普通类：p = Person("张三", "不是数字")  ← 不报错
# Pydantic：p = Person(name="张三", age="不是数字")  ← 直接报错！
```

### 核心模型逐个看

```python
# ---- 1. 聊天的基本单位：一条消息 ----
class ChatMessage(BaseModel):
    role: str = Field(description="角色：system | user | assistant")
    # system = 系统指令, user = 用户说的, assistant = AI 回复的
    content: str = Field(description="文本内容")

# 使用示例：
# msg = ChatMessage(role="user", content="你好")


# ---- 2. 普通聊天请求 ----
class ChatRequest(BaseModel):
    messages: list[ChatMessage] = Field(min_length=1)
    # ↑ 消息列表，至少 1 条。比如：
    # [{"role":"user","content":"你好"}]
    model: str | None = None          # 不指定就用默认的
    temperature: float = 0.7          # 随机性：0=死板，1=狂野
    max_tokens: int | None = None     # 回复最大长度
    conversation_id: str | None = None  # 会话 ID（用于记忆）


# ---- 3. Agent 模式请求（比普通请求多一个 agent_mode） ----
class AgentChatRequest(BaseModel):
    messages: list[ChatMessage] = Field(min_length=1)
    model: str | None = None
    temperature: float = 0.7
    conversation_id: str | None = None
    agent_mode: str = "react"
    # ↑ 关键差异！"react" = ReAct 模式，"plan_execute" = 先规划再执行


# ---- 4. Agent 模式响应（比普通响应多很多字段） ----
class AgentChatResponse(BaseModel):
    id: str                           # 响应 ID
    model: str                        # 用了哪个模型
    content: str                      # AI 的回复文字
    trace_id: str | None = None       # 追踪 ID
    usage: dict | None = None         # Token 用量

    # === Agent 专属字段 ===
    agent_mode: str = "react"         # 最终用了哪种模式
    intent: IntentInfo | None = None  # 意图识别结果
    steps: list[StepInfo] = []        # ★ ReAct 执行步骤（前端拿这个画步骤面板）
    tool_calls: list[str] = []        # 本次调用了哪些工具
    reflection_score: int | None = None  # 反思评分 0-100
    degraded: bool = False            # 是否失败降级


# ---- 5. ReAct 步骤信息（前端步骤面板的数据源） ----
class StepInfo(BaseModel):
    step: int                         # 第几步
    phase: str = "react"              # 阶段标识
    thought: str | None = None        # AI 当时的思考过程
    action: str | None = None         # 调用了哪个工具
    action_input: dict | None = None  # 工具的输入参数
    observation: str | None = None    # 工具返回的结果
    final_answer: str | None = None   # 最终答案
    error: str | None = None          # 出错信息


# ---- 6. RAG 检索结果 ----
class RetrievalResult(BaseModel):
    id: str                           # 文档块 ID
    content: str                      # 文档块内容
    score: float = 0.0                # 相关性分数
    source: str = "vector"            # 来源：vector/keyword/hybrid


# ---- 7. 引用标注 ----
class Citation(BaseModel):
    index: int                        # 引用编号 [1]
    result_id: str                    # 对应文档块 ID
    snippet: str                      # 文档片段预览
```

### 为什么用 Pydantic

1. **自动校验**：`temperature: float = Field(ge=0.0, le=2.0)` — 超出范围直接返回错误
2. **自动文档**：FastAPI 从 Pydantic 模型生成 Swagger 文档
3. **序列化**：`.model_dump()` 一键转字典，`.model_dump_json()` 一键转 JSON 字符串

## app/core/agent/react_agent.py — ★ AI 的「大脑」

**这是项目最重要的文件，没有之一。** 它实现了让 AI 像人类一样「边思考边行动」的循环。如果只看一个文件，就看这个。

### 先理解：什么是 ReAct

```
人类解决问题的方式：
  想 → 做 → 看结果 → 再想 → 再做 → 直到解决

AI 用 ReAct 模仿这个过程：
  Thought → Action → Observation → Thought → Action → Observation → ... → Final Answer
  思考     行动      观察         思考     行动      观察                最终答案
```

### 逐行白话解释

```python
# === 第 1 部分：System Prompt（系统指令） ===

REACT_SYSTEM_PROMPT = """你是一个严谨的智能助手，必须使用 ReAct 方式回答问题。

## 输出格式（严格遵守）
### 若需要调用工具：
Thought: <为什么要调工具>
Action: <工具名>
Action Input: <JSON 格式的参数>

### 若可直接作答：
Thought: <依据>
Final Answer: <完整答案>

## 规则
- 不要编造工具名
- Action Input 必须是合法 JSON
- 若工具返回错误，分析原因并调整
"""
# 这个 prompt 教 AI 按照固定格式输出，然后代码用正则解析这个格式


# === 第 2 部分：解析函数（把 AI 的文字输出拆成结构化数据）====

def _parse_react_step(text: str) -> Dict:
    """
    输入：AI 的一段文字，比如：
        "Thought: 我需要搜索
         Action: web_search
         Action Input: {"query": "世界杯冠军"}"

    输出：结构化字典：
        {"thought": "我需要搜索",
         "action": "web_search",
         "action_input": {"query": "世界杯冠军"}}
    """

    # 1. 提取 Thought（用正则匹配 "Thought: xxx"）
    thought_m = re.search(
        r"Thought:\s*(.+?)(?=\n(?:Action:|Final Answer:)|\Z)",
        text, re.S | re.I
    )
    # 解释正则：
    # Thought:        → 匹配文字 "Thought:"
    # \s*             → 跳过空格
    # (.+?)           → 捕获任意内容（非贪婪，直到碰到后面的标记）
    # (?=\n(?:Action:|Final Answer:)|\Z)
    #                  → 前瞻断言：遇到 "Action:" 或 "Final Answer:" 或文本结束 就停

    # 2. 检查是否是 Final Answer（最终答案）
    if re.search(r"Final Answer:\s*", text, re.I):
        fa_m = re.search(r"Final Answer:\s*(.+)\Z", text, re.S | re.I)
        out["final_answer"] = fa_m.group(1).strip()
        out["done"] = True       # ← 标记完成！循环终止
        return out

    # 3. 提取 Action（工具名）
    action_m = re.search(r"Action:\s*(\S+)", text, re.I)
    # Action:\s*  → 匹配 "Action:" + 空格
    # (\S+)       → 捕获非空格字符（工具名，如 "web_search"）

    # 4. 提取 Action Input（JSON 参数）
    input_m = re.search(r"Action Input:\s*(\{[\s\S]*\})", text)
    # Action Input:\s*  → 匹配 "Action Input:" + 空格
    # (\{[\s\S]*\})     → 捕获花括号及其中的所有内容（即 JSON）
    if input_m:
        out["action_input"] = json.loads(input_m.group(1))
        # json.loads 把字符串转成 Python 字典

    # 5. 宽松兜底：如果严格 JSON 解析失败，尝试更宽松的匹配
    else:
        loose = re.search(r"Action Input:\s*([\s\S]+?)(?=\n\n|\Z)", text, re.I)
        if loose:
            try:
                out["action_input"] = json.loads(loose.group(1).strip())
            except json.JSONDecodeError:
                out["parse_error"] = "无法解析 Action Input"

    return out


# === 第 3 部分：ReAct 主循环 ===

class ReActAgent:
    """ReAct 循环的核心类"""

    def __init__(self, llm, tools, memory, max_steps=10, session_id=None):
        self._llm = llm           # LLM 接口（能调用 AI 的对象）
        self._tools = tools        # 工具接口（能执行工具的对象）
        self._memory = memory      # 记忆接口（能存取记忆的对象）
        self.max_steps = max(1, max_steps)  # 最多几步（防止死循环）

    def _tool_catalog_text(self, tool_names, tool_descriptions=""):
        """生成工具列表文本，写入 System Prompt"""
        if tool_descriptions:
            return tool_descriptions  # 有描述就用描述（修复后）
        # 兜底：只列工具名
        return "\n".join(f"- {name}" for name in tool_names)

    async def run(self, query: str, context: dict) -> AgentResult:
        """
        执行 ReAct 循环。
        query: 用户的问题
        context: 包含 tool_names, session_id, trace_callback 等
        """

        # ── 准备阶段 ──
        tool_names = list(context.get("tool_names") or [])
        tool_descriptions = str(context.get("tool_descriptions", ""))
        session_id = str(context.get("session_id", "default"))

        # 读取记忆（之前对话的摘要）
        mem_snippets = []
        if self._memory:
            try:
                mem_snippets = await self._memory.get_relevant(session_id, query, limit=8)
            except:
                logger.warning("读取记忆失败，继续执行")  # 不崩溃

        history_lines = []  # 记录每个步骤的过程

        # ── 主循环：最多 max_steps 步 ──
        for step_idx in range(self.max_steps):

            # 第 1 步：构造给 AI 的完整消息
            tool_desc = self._tool_catalog_text(tool_names, tool_descriptions)
            user_prompt = build_react_user_prompt(
                query,                    # 用户问题
                tool_desc,                # 可用工具列表
                "\n".join(history_lines)  # 之前步骤的记录
            )

            # 组装最终的消息列表（OpenAI 格式）
            messages = [
                {"role": "system", "content": REACT_SYSTEM_PROMPT + memory_context},
                {"role": "user", "content": user_prompt},
            ]

            # 第 2 步：调用 AI
            try:
                raw = await self._llm.acomplete(messages, temperature=0.2)
                # temperature=0.2：很低的随机性，让 AI 严格遵循格式
            except Exception as e:
                return AgentResult(success=False, error=f"LLM 调用失败: {e}")

            # 第 3 步：解析 AI 的输出
            parsed = _parse_react_step(raw)

            # 第 4 步：如果是 Final Answer → 结束循环
            if parsed.get("done") and parsed.get("final_answer"):
                answer = parsed["final_answer"]
                # 写入记忆
                await self._memory.append_turn(session_id, "assistant", answer)
                return AgentResult(success=True, final_answer=answer)

            # 第 5 步：否则执行工具
            action = parsed.get("action")           # 比如 "web_search"
            action_input = parsed.get("action_input")  # 比如 {"query":"世界杯"}

            if not action:
                # AI 既没给 Final Answer 也没给 Action → 无法继续
                return AgentResult(success=False, error="未解析到 Action 或 Final Answer")

            # 安全检查：工具名必须在允许列表里
            if tool_names and action not in tool_names:
                obs = f"错误：工具 [{action}] 不在允许列表中。"
            else:
                # 执行工具
                try:
                    obs = await self._tools.invoke(action, action_input or {})
                except Exception as e:
                    obs = f"工具执行异常: {e}"

            # 第 6 步：把 Observation 写入历史，继续下一轮
            history_lines.append(
                f"Step {step_idx+1}\n"
                f"Thought: {parsed.get('thought', '')}\n"
                f"Action: {action}\n"
                f"Observation: {obs}\n"
            )
            # 下一轮循环时，AI 会看到这些历史，知道之前做了什么

        # 循环结束但没拿到 Final Answer（达到最大步数）
        return AgentResult(success=False, error=f"已达到最大步数限制({self.max_steps})")
```

### ReAct 执行示例

```
用户问: "2026年世界杯冠军是谁？"

Step 1:
  Thought: "这个问题需要实时信息，我应该搜索"
  Action: web_search
  Action Input: {"query": "2026年世界杯冠军"}
  → Observation: "搜索结果：2026年世界杯...葡萄牙队1-0击败阿根廷..."

Step 2:
  Thought: "已获得搜索结果，可以给出答案"
  Final Answer: "2026年世界杯冠军是葡萄牙队，他们在决赛中1-0战胜了阿根廷队。"
```

### 面试要点

- **为什么不用 LangChain 的 ReAct？** 自研实现让你完全掌控每一步的行为，而且面试时能解释底层原理
- **max_steps 为什么是 10？** 防止死循环。实际使用中大多数问题 2-5 步就搞定了
- **为什么用正则解析而不是 function calling？** 正则解析更灵活（不依赖特定 API 格式），但确实更脆弱。生产环境建议升级为 function calling
- **temperature=0.2 的原因**：需要 AI 严格遵循输出格式，太高的随机性会导致格式错误无法解析

## app/core/agent/planner.py — AI 的「计划表」

**作用**：另一种 AI 工作模式。先把大任务拆成小步骤，再一步步执行。比 ReAct 更适合复杂的多步任务。

### 什么是 Plan-and-Execute

```
ReAct（边想边做）:
  想 → 做 → 看 → 想 → 做 → 看 → ...

Plan-and-Execute（先规划再执行）:
  先做好完整计划 → 第1步 → 第2步 → 第3步 → 汇总
     ↓ 如果某步失败
  重新规划 → 第1步 → 第2步' → 汇总
```

### 逐行白话解释

```python
# === 第 1 部分：Plan 阶段（生成计划）===

PLAN_SYSTEM_PROMPT = """你是任务规划专家。请将用户目标拆分为有序、可执行的子任务列表。

## 输出格式（仅输出 JSON）
{
  "subtasks": [
    {
      "id": "t1",
      "title": "子任务标题",
      "description": "具体要做什么",
      "action_type": "tool|reasoning",
      "tool_name": "若 action_type 为 tool 则填写工具名，否则 null",
      "tool_args_hint": "工具参数要点"
    }
  ]
}

规则：
- action_type 为 reasoning 表示主要靠 AI 推理（如整理信息、写总结）
- action_type 为 tool 表示需要调用工具（如搜索、计算）
- 子任务数量建议 2～8 个
"""

class PlannerAgent:
    async def plan(self, query: str) -> List[SubTask]:
        """根据用户目标，让 AI 生成子任务列表"""
        messages = [
            {"role": "system", "content": PLAN_SYSTEM_PROMPT},
            {"role": "user", "content": f"用户目标：\n{query}\n\n请输出 JSON 计划。"},
        ]
        try:
            raw = await self._llm.acomplete(messages, temperature=0.3)
            data = json.loads(raw)  # 解析 JSON
            return [SubTask(id=item["id"], title=item["title"], ...) for item in data["subtasks"]]
        except:
            # ★ 降级 1：如果计划生成失败，退回单步推理
            return [SubTask(id="fallback_1", title="直接回答", action_type="reasoning")]


# === 第 2 部分：Execute 阶段（执行计划）===

    async def execute(self, plan, query, session_id, tool_names, trace_callback):
        """逐个执行子任务"""

        for idx, task in enumerate(plan):
            if task.action_type == "tool" and task.tool_name:
                # tool 类型 → 调用工具
                args = {"hint": task.tool_args_hint or "", "user_query": query}
                obs = await self._tools.invoke(task.tool_name, args)
                # 记录 Observation
            else:
                # reasoning 类型 → 让 AI 基于已有结果推理
                msgs = [
                    {"role": "system", "content": "你是执行专家..."},
                    {"role": "user", "content": f"原始问题：{query}\n子任务：{task.title}\n已有结果：{ctx}"},
                ]
                text = await self._llm.acomplete(msgs, temperature=0.3)

            # 如果某步失败 → 返回错误，触发 replan
            if 失败:
                return AgentResult(success=False, error=str(e))

        # 所有子任务完成 → 让 AI 汇总成最终答案
        final = await self._llm.acomplete([
            {"role": "system", "content": "你是总结助手..."},
            {"role": "user", "content": f"问题：{query}\n执行记录：{results}"},
        ])
        return AgentResult(success=True, final_answer=final)


# === 第 3 部分：Replan 阶段（失败重规划）===

    async def replan(self, plan, results, error):
        """根据错误信息和已完成结果，让 AI 重新规划"""
        payload = {
            "previous_plan": [...],    # 原来的计划
            "results_so_far": [...],   # 已完成的结果
            "error": error,            # 失败原因
        }
        messages = [
            {"role": "system", "content": REPLAN_SYSTEM_PROMPT},
            {"role": "user", "content": json.dumps(payload)},
        ]
        raw = await self._llm.acomplete(messages, temperature=0.3)
        return _parse_subtasks(json.loads(raw))


# === 第 4 部分：高层封装 ===

    async def run_with_replan(self, query, session_id, tool_names, trace_callback):
        """完整流程：计划 → 执行 → 失败→重规划→再执行 → ..."""
        current_plan = await self.plan(query)

        for attempt in range(self.max_replan_attempts + 1):
            res = await self.execute(current_plan, ...)
            if res.success:
                return res  # ✓ 成功！

            # 失败 → 重规划，下次循环再执行
            current_plan = await self.replan(current_plan, res.steps, res.error)

        return AgentResult(success=False, error="执行失败且达到最大重规划次数")
```

### 什么时候用 ReAct，什么时候用 Plan-Execute？

| 场景 | 推荐模式 | 原因 |
|------|---------|------|
| 简单问答 | ReAct | 不需要规划 |
| 需要查资料 | ReAct | 搜索→回答，2 步搞定 |
| 多步数据分析 | Plan-Execute | 先规划步骤再执行 |
| 写报告 | Plan-Execute | 需要收集信息→分析→撰写 |

## app/core/agent/orchestrator.py — AI 的「指挥官」

**作用**：统管全局。决定用 ReAct 还是 Plan-Execute，管理记忆和追踪，失败时自动切换策略。

### 逐行白话解释

```python
class AgentOrchestrator:
    """
    编排器的职责：
    1. 决定用哪种 Agent 模式（ReAct / Plan-Execute）
    2. 管理整个执行过程的记忆和追踪
    3. Plan 失败 → 自动降级到 ReAct
    4. 可选地对答案做反思质检
    """

    def __init__(self, config, model_router, memory_manager, tool_registry, tracer):
        self._enable_reflection = config.get("enable_reflection", True)
        self._fallback_on_plan_failure = config.get("fallback_react_on_plan_failure", True)
        # ↑ 配置控制：Plan 失败是否自动降级到 ReAct

    async def run(self, user_input, session_id, mode, intent):
        """
        主入口。mode 可以是 "react" 或 "plan_execute"
        """

        # ── 第 1 步：选模式 ──
        if mode == "plan_execute":
            result = await self._run_plan_execute(...)
            # ★ 如果 Plan 失败了且允许降级 → 换 ReAct 再试
            if not result.success and self._fallback_on_plan_failure:
                degraded = True
                result = await self._run_react(...)  # 降级！
        else:
            result = await self._run_react(...)

        # ── 第 2 步：反思质检（可选）──
        if self._enable_reflection and answer:
            try:
                reflection_report = await self._run_reflection(...)
                # reflection_report.quality_score    ← 0-100 评分
                # reflection_report.likely_hallucination ← 是否疑似幻觉
            except:
                pass  # 反思失败不影响主流程

        # ── 第 3 步：返回结果 ──
        return AgentResponse(
            answer=answer,
            mode_used=mode_used,   # 最终用了哪种模式
            steps=steps,           # 执行步骤（前端面板用）
            reflection=reflection, # 反思报告
            degraded=degraded,     # 是否降级
        )

# 依赖反转：编排器不 import 具体类，只依赖「协议」
# 这意味着你可以随时替换底层实现，编排器代码不用改
class ModelRouter(Protocol):
    def get_llm(self, purpose: str) -> Any: ...
    # purpose 可以是 "react"、"planner"、"reflection"
    # 不同用途可能用不同模型（如 cheap model 做反思，powerful model 做主流程）

class MemoryManager(Protocol):
    async def get_relevant(self, session_id, query, limit) -> List[str]: ...
    async def append_turn(self, session_id, role, content, metadata) -> None: ...

class Tracer(Protocol):
    def new_trace_id(self) -> str: ...
    def start_span(self, name, trace_id, attributes) -> Any: ...
```

### 降级执行流程图

```
用户问: "帮我写一份数据分析报告"
    ↓
编排器: mode = "plan_execute"
    ↓
Planner: 生成计划 [t1:查数据, t2:计算, t3:分析, t4:写报告]
    ↓
Execute: t1 ✓ → t2 ✗（计算失败）
    ↓
Replan:  新计划 [t2':估算, t3':简化分析]
    ↓
Execute: t2' ✗（又失败）
    ↓
编排器: "Plan 全失败了，换 ReAct 试试"
    ↓
ReAct:   直接让 AI 基于搜索到的信息写报告
    ↓
返回结果（degraded=True） ← 前端看到 degraded 标记
```

## app/core/agent/reflection.py — AI 的「质检员」

**作用**：用一个独立的 AI 调用，给主 AI 的答案打分、检查有没有幻觉。

### 逐行白话解释

```python
REFLECTION_SYSTEM_PROMPT = """你是严格的输出质量审查员。

## 你必须只输出一个 JSON 对象：
{
  "quality_score": 0-100 的整数,     ← 综合评分
  "is_complete": true/false,         ← 回答完整吗？
  "likely_hallucination": true/false, ← 怀疑幻觉吗？
  "hallucination_reasons": ["疑点1", "疑点2"],
  "completeness_notes": "是否遗漏要点",
  "suggestions": ["建议1", "建议2"],     ← 给助手的改进建议
  "summary": "一句中文总结"
}

审查标准：
- 幻觉：回答是否包含无依据的具体事实
- 完整性：是否覆盖用户问题的核心子问题
- 质量分：综合考虑正确性、清晰度与有用性
"""

class ReflectionAgent:
    def __init__(self, llm, min_quality_to_pass=60):
        self._llm = llm
        self.min_quality_to_pass = 60  # 低于 60 分要告警

    async def reflect(self, user_query, agent_answer, evidence_snippets, trace_summary):
        """对答案进行审查"""

        # 构造审查消息
        messages = [
            {"role": "system", "content": REFLECTION_SYSTEM_PROMPT},
            {"role": "user", "content": f"""
                ## 用户问题
                {user_query}

                ## 助手回答
                {agent_answer}

                ## 外部证据/观察片段
                {evidence_snippets or "（无外部证据）"}

                请输出 JSON 审查结果。
            """},
        ]

        try:
            raw = await self._llm.acomplete(messages, temperature=0.1)  # 最低温度
            data = json.loads(raw)
            return ReflectionReport(
                quality_score=clamp(data["quality_score"], 0, 100),
                is_complete=data["is_complete"],
                likely_hallucination=data["likely_hallucination"],
                suggestions=data["suggestions"],
                summary=data["summary"],
            )
        except:
            # 审查失败了 → 返回保守评分（50 分，建议人工复核）
            return ReflectionReport(
                quality_score=50,
                is_complete=False,
                suggestions=["请人工复核该回答"],
                summary="反思流程异常，已降级为保守评分",
            )

    def should_retry_or_warn(self, report):
        """根据评审报告决定是否重试或告警"""
        return {
            "warn_low_quality": report.quality_score < 60,           # 质量分低于 60
            "suggest_retry": report.likely_hallucination or not report.is_complete,
            # ↑ 幻觉或不完整 → 建议重试
        }
```

### 面试要点

**为什么用独立 LLM 审查，而不是同一个 LLM 自查？**
因为 LLM 很难发现自己的错误（类似自己检查自己的作业）。用独立调用+更低的 temperature+不同的 prompt 角色，审查更客观。

```python
# ===== 1. 嵌入模型：把文字变成数字向量 =====

def _get_embedding_model():

        # 语义相近的句子，向量也相近

# 示例：
# "今天天气真好"  → [0.12, -0.34, 0.56, ..., 0.08]  (384 个数)
# "天气不错"      → [0.11, -0.32, 0.54, ..., 0.07]  (和上面很接近！)
# "苹果手机多少钱" → [-0.45, 0.78, -0.23, ..., 0.15] (和上面差很远)


# ===== 2. BM25 关键词检索（自研实现，不用任何库）=====

class _BM25:
    """
    BM25 是经典的关键词检索算法。
    核心思想：
    - 词在越少的文档中出现 → 这个词越重要（IDF 权重高）
    - 词在同一文档中出现多次 → 这个文档越相关（TF 权重高）
    - 但出现太多也没意义（TF 饱和，k1 控制）
    - 长文档中天然词频高 → 要做长度归一化（b 控制）
    """

    def __init__(self, k1=1.5, b=0.75):
        self.k1 = k1  # 词频饱和度参数（默认 1.5）
        self.b = b    # 文档长度归一化（默认 0.75）

    def add(self, doc_id, text):
        """添加一篇文档到索引"""
        tokens = _tokenize(text)  # 分词：把"今天天气真好"→["今天","天气","真好"]

        # 统计本文档中每个词出现几次
        tf = {}
        for t in tokens:
            tf[t] = tf.get(t, 0) + 1

        # 更新全局：每个词在多少个文档中出现过
        for t in tf:
            self._df[t] = self._df.get(t, 0) + 1

        # 重新计算所有词的 IDF（逆文档频率）
        self._idf = {}
        for t, df in self._df.items():
            # IDF = log(1 + (N - df + 0.5) / (df + 0.5))
            # 如果 df=1（只在 1 篇文档出现），IDF 很大 → 重要
            # 如果 df=N（每篇都有），IDF 很小 → 不重要（如"的""了"）
            self._idf[t] = math.log(1.0 + (N - df + 0.5) / (df + 0.5))

    def search(self, query, top_k):
        """搜索最相关的 top_k 篇文档"""
        q_terms = _tokenize(query)

        for i in range(N):  # 遍历每一篇文档
            tf = self._doc_freqs[i]   # 本文档的词频
            dl = self._doc_lens[i]    # 本文档长度
            s = 0.0
            for term in q_terms:
                idf = self._idf.get(term, 0.0)  # 这个词的 IDF
                f = tf.get(term, 0)             # 这个词在本文档出现几次
                if f == 0:
                    continue
                # BM25 核心公式：
                # score = IDF × (f × (k1+1)) / (f + k1×(1-b + b×dl/avgdl))
                s += idf * (f * (k1+1)) / (f + k1*(1-b + b*dl/avgdl))
            scores.append((doc_id, s, content))

        return sorted by score desc[:top_k]  # 分数最高的 top_k 篇


# ===== 3. RRF 混合检索融合 =====

def hybrid_search(self, query_vec, query_text, top_k):
    """向量检索 + BM25 → RRF 融合"""
    vec_results = self.vector_search(query_vec, top_k * 2)  # 召回 2 倍
    kw_results = self.keyword_search(query_text, top_k * 2)

    rrf_k = 60  # RRF 常数

    # 向量检索：排第 1 的贡献 = 1/(60+1+1) = 1/62
    #          排第 2 的贡献 = 1/(60+2+1) = 1/63
    for rank, r in enumerate(vec_results):
        scores[r.id] = 1.0 / (rrf_k + rank + 1)

    # BM25 检索：同样逻辑
    for rank, r in enumerate(kw_results):
        scores[r.id] += 1.0 / (rrf_k + rank + 1)
        # ↑ 两路都命中的文档分数会累加，排名更靠前

    # 按融合后的总分排序
    return sorted(scores, key=lambda x: -scores[x])[:top_k]


# ===== 4. 四级降级设计 =====

class RAGService:
    def __init__(self):
        # 嵌入模型降级
        try:      self._embed = SentenceTransformer("all-MiniLM-L6-v2")  # 384维语义
        except:   self._embed = _NGramEmbedding()                       # 256维字符哈希

        # 向量存储降级
        try:      self._store = _MilvusRAGStore("milvus.db")            # Milvus Lite

        # 啥都没有 → 256 维字符哈希 + 暴力搜索（慢但能跑）
```

### 面试要点

**为什么要 BM25 + 向量两种检索？**
- 向量检索：能找到意思相近但用词不同的内容（"汽车"≈"轿车"）
- BM25：对专有名词、编码、型号等精确匹配更好（"iPhone 15"不会模糊成"iPhone 14"）
- 两者互补，RRF 融合取长补短

```python
class MultiRetriever:
    """统一检索入口"""

    async def retrieve(self, query, top_k, mode="hybrid"):
        if mode == "vector":
            return await self.vector_search(query, top_k)
        elif mode == "keyword":
            return await self.keyword_search(query, top_k)
        else:
            return await self.hybrid_search(query, top_k)

    async def hybrid_search(self, query, top_k):
        """关键：并行执行两路检索，用 asyncio.gather"""
        vec_task = self.vector_search(query, top_k)
        kw_task = self.keyword_search(query, top_k)

        # 比串行快很多（假如两路各耗时 100ms，并行就是 100ms 而非 200ms）
        vec_res, kw_res = await asyncio.gather(vec_task, kw_task)
        return self._rrf_fuse([vec_res, kw_res], top_k)
```

## app/core/rag/reranker.py — 搜索结果的「精排员」

**作用**：向量检索输出粗筛结果后，用更精确的模型重新排序。

```python
class Reranker:
    """
    为什么需要重排序？
    - 向量检索用 Bi-Encoder（分别编码 query 和 doc 再算相似度） → 快但粗糙
    - 重排序用 Cross-Encoder（把 query+doc 一起喂给模型打分） → 慢但精准
    - 组合使用：粗筛 20 条 → 重排取 top 5
    """

    def _load_model(self):
        """懒加载 CrossEncoder 模型（ms-marco-MiniLM-L-6-v2）"""
        if self._model is None:
            from sentence_transformers import CrossEncoder
            self._model = CrossEncoder(self._model_name)
        return self._model

    async def rerank(self, query, documents, top_k=5):
        model = self._load_model()
        # 构造 (query, document) 对
        pairs = [(query, d.content) for d in documents]
        # CrossEncoder 直接预测相关性分数
        scores = await asyncio.to_thread(model.predict, pairs)
        # 按分数降序排列，取 top_k
        ranked = sorted(zip(documents, scores), key=lambda x: -x[1])
        return [doc for doc, score in ranked[:top_k]]
```

## app/core/memory/short_term.py — 记忆的「草稿纸」

**作用**：用 Redis 存储最近对话。当对话太长时，把旧的压缩成摘要。

### 滑动窗口 + 自动压缩

```
初始状态：[msg1, msg2, msg3, ..., msg18, msg19, msg20]  ← 20 条，刚好

第 21 条进来 → 触发压缩：
  旧的：[msg1, msg2, ..., msg10]  → LLM 摘要："用户询问了...AI 回答了..."
  保留：[msg11, msg12, ..., msg21]  ← 最近 10 条保留原文

压缩后：[摘要消息, msg11, msg12, ..., msg21]  ← 12 条，空出 8 条空间
```

### 逐行白话解释

```python
class ShortTermMemory:
    def __init__(self, redis_client, llm, window_size=20, max_tokens=4000):
        self._redis = redis_client
        self._llm = llm           # 用于生成摘要的 LLM
        self.window_size = 20     # 最多保留多少条
        self.max_tokens = 4000    # Token 上限
        self._encoding = tiktoken.get_encoding("cl100k_base")
        # tiktoken 是 OpenAI 的 token 计数库，精确计算 token 数
        # 中文字符约 1.5-2 token/字，英文约 0.75 token/词

    def _count_tokens(self, text):
        """计算文本的 token 数"""
        return len(self._encoding.encode(text))

    async def _compress_if_needed(self, session_id):
        """如果消息太多或 token 太大，触发压缩"""
        n = await self._redis.llen(key)        # 当前消息数
        msgs = await self.get_history(session_id)
        total_tokens = sum(self._count_tokens(m.content) for m in msgs)

        # 两个条件都没超 → 不压缩
        if n <= self.window_size and total_tokens <= self.max_tokens:
            return

        # 保留最近 10 条，旧的发给 LLM 压缩
        keep = max(2, self.window_size // 2)  # 10
        to_summarize = msgs[:-keep]  # 旧的
        tail = msgs[-keep:]          # 保留的

        # 让 LLM 把旧消息压缩成摘要
        summary_text = await self._summarize_messages(to_summarize)

        # 替换 Redis 中的内容
        await self._redis.delete(key)
        summary_msg = Message(role="system", content=f"[历史摘要]\n{summary_text}")
        await self._redis.rpush(key, summary_msg)  # 先放摘要
        for m in tail:
            await self._redis.rpush(key, m)        # 再放最近的消息

    async def _summarize_messages(self, messages):
        """让 LLM 把消息列表压缩成摘要"""
        prompt = "请将以下对话压缩为简洁中文摘要，保留关键事实与用户意图：\n\n"
        prompt += "\n".join(f"{m.role}: {m.content}" for m in messages)
        raw = await self._llm.ainvoke(prompt)
        return raw.content.strip()
```

## app/core/memory/long_term.py — 记忆的「档案柜」

**作用**：用向量库存储"重要记忆"，以后可以按语义搜索。

```python
class LongTermMemory:
    """
    与短期记忆的区别：
    - 短期记忆：最近 N 条对话（时间窗口）
    - 长期记忆：重要的、值得记住的内容（语义搜索）

    使用场景：
    - 一个月前用户说自己喜欢 Python → 现在问"推荐什么编程语言" → 召回这条记忆
    """

    async def store(self, session_id, content, metadata):
        """存入一条长期记忆"""
        vec = self._embed.embed_query(content)  # 文字 → 向量
        self._coll.insert([{
            "pk": memory_id,
            "embedding": vec,        # 向量（用于相似度搜索）
            "content": content,      # 原文
            "session_id": session_id, # 属于哪个会话
        }])
        self._coll.flush()  # 刷盘（确保写入）

    async def recall(self, query, session_id, top_k=5):
        """按语义搜索最相关的记忆"""
        vec = self._embed.embed_query(query)  # 查询 → 向量
        # 在指定 session_id 下搜索最相似的记忆
        results = self._coll.search(
            data=[vec],
            expr=f'session_id == "{session_id}"',  # Milkvus 过滤表达式
            limit=top_k,
        )
        return [MemoryItem(id=..., content=..., score=...) for hit in results]
```

## app/core/persistent_memory.py — ★ 记忆的「三保险」

**这是项目最巧妙的设计之一。** 三级存储，任何一层挂了都不影响。

### 数据流

```
写入 append_turn():
    ├─ 1. _InMemoryMemoryManager  ← Python dict，100% 成功
    ├─ 2. SessionRepository       ← PostgreSQL 持久化
    └─ 3. ShortTermMemory         ← Redis 热路径

读取 get_relevant():
    ├─ 1. ShortTermMemory (Redis)    ← 最快，命中率高
    ├─ 2. SessionRepository (PG)     ← Redis 挂了用这个
    └─ 3. _InMemoryMemoryManager     ← PG 也挂了用这个（最终兜底）
```

### 逐行白话解释

```python
class PersistentMemoryManager:
    def __init__(self, redis_client, compress_llm, pg_repo):
        self._pg = pg_repo  # PostgreSQL 仓库

        # Redis：如果连接正常，初始化 ShortTermMemory
        self._redis_ok = redis_client is not None
        if self._redis_ok and compress_llm:
            try:
                self._redis_stm = ShortTermMemory(redis_client, compress_llm)
            except:
                self._redis_ok = False  # ← 标记失败，后续不走 Redis

        # 内存降级：始终可用
        self._fallback = _InMemoryMemoryManager()

    async def get_relevant(self, session_id, query, limit=8):
        """三层读取：Redis → PG → 内存"""
        # 第 1 层：Redis
        if self._redis_ok:
            try:
                return await self._redis_stm.get_relevant(session_id, query, limit)
            except:
                self._redis_ok = False  # ← 标记失败，下次直接跳过

        # 第 2 层：PG
        if self._pg_ok:
            try:
                return await self._pg.get_recent_messages(session_id, limit)
            except:
                self._pg_ok = False

        # 第 3 层：内存（永不失败）
        return await self._fallback.get_relevant(session_id, query, limit)

    async def append_turn(self, session_id, role, content, metadata=None):
        """写入：三层全写"""
        # 内存兜底：始终写入
        await self._fallback.append_turn(session_id, role, content, metadata)

        # PG 主路径
        if self._pg_ok:
            try:
                await self._pg.add_message(session_id, role, content, metadata)
            except:
                pass  # 写入失败不阻塞

        # Redis 辅助路径
        if self._redis_ok and self._redis_stm:
            try:
                await self._redis_stm.append_turn(session_id, role, content, metadata)
            except:
                pass  # 同上
```

```python
        """遍历意图树，统计每个分支的关键词命中数"""

        # "搜索"命中 → 任务/搜索, hits=1
        # confidence = min(0.95, 0.45 + 0.12 * 1) = 0.57

    async def clarify(self, query, intent_result):
        """置信度太低时，生成澄清提示"""
        if intent_result.confidence >= 0.55:
            return ""  # 够高了，不需要澄清
        return "我不太确定您的具体需求。您是想了解「问答」相关内容吗？请补充更多信息。"
```

## app/infrastructure/llm/circuit_breaker.py — ★ LLM 的「保险丝」

**这是项目最精巧的工程模块。** 128 行纯 Python，实现了电气工程中「保险丝」的概念。

### 先理解：为什么需要熔断器？

```
正常情况：
  App → LLM API → 返回结果 ✓

故障情况：
  App → LLM API → 超时 → 重试 → 又超时 → 再重试 → ...  → 系统卡死

有熔断器：
  App → 熔断器 → 失败了 5 次 → 熔断！→ 接下来 60 秒直接拒绝，不再浪费资源
  → 60 秒后 → 试探一下 → 成功了 → 恢复 ✓
            → 还是失败 → 继续熔断
```

### 逐行白话解释

```python
class CircuitState(str, Enum):
    CLOSED = "closed"        # 正常模式（绿灯）
    OPEN = "open"            # 熔断模式（红灯）
    HALF_OPEN = "half_open"  # 试探模式（黄灯）

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=60.0, half_open_max=3):
        """
        failure_threshold=5  : 连续失败 5 次就熔断
        recovery_timeout=60  : 熔断后等 60 秒再试试
        half_open_max=3      : 试探模式最多试 3 次
        """
        self._state = CircuitState.CLOSED  # 初始：正常
        self._failure_count = 0            # 失败计数

    async def call(self, func, *args, **kwargs):
        """
        用熔断器包裹一个异步函数调用。
        用法：
            breaker = CircuitBreaker()
            result = await breaker.call(llm_client.chat, messages)
        """
        async with self._lock:  # 加锁，保证并发安全
            # ── OPEN 状态：检查是否该试探了 ──
            if self._state == CircuitState.OPEN:
                if time.monotonic() - self._last_failure_time >= 60:
                    self._state = CircuitState.HALF_OPEN  # → 黄灯
                    self._half_open_attempts = 0
                else:
                    raise RuntimeError("熔断器 OPEN，拒绝调用")

            # ── HALF_OPEN 状态：限制试探次数 ──
            if self._state == CircuitState.HALF_OPEN:
                if self._half_open_attempts >= 3:
                    raise RuntimeError("试探次数已达上限")
                self._half_open_attempts += 1

        # ── 执行真正的调用 ──
        try:
            result = await func(*args, **kwargs)
        except Exception:
            async with self._lock:
                self._record_failure()   # 失败数+1，可能触发 OPEN
            raise

        async with self._lock:
            self._record_success()       # 成功 → 归零，回到 CLOSED

        return result

    def _record_failure(self):
        self._failure_count += 1
        if self._state == CircuitState.HALF_OPEN:
            # 黄灯试探失败 → 立刻回到红灯（不等 60s）
            self._state = CircuitState.OPEN
        elif self._failure_count >= 5:
            # 失败 5 次 → 跳闸！
            self._state = CircuitState.OPEN

    def _record_success(self):
        # 成功 → 重置一切
        self._state = CircuitState.CLOSED
        self._failure_count = 0
```

### 面试要点

- **为什么用 `time.monotonic()` 而不是 `time.time()`？**
  `time.time()` 可能因为系统时间调整（NTP 同步）而回退，导致熔断器意外提前恢复。`time.monotonic()` 保证单调递增。

- **为什么 HALF_OPEN 试探失败要立即回 OPEN？**
  因为试探证明服务还没恢复，没必要再等一个超时周期。

## app/infrastructure/llm/model_router.py — LLM 的「接线员」

**作用**：管理多个 AI 模型，按优先级选择，一个不行自动换下一个。

```python
class ModelRouter:
    def __init__(self, model_configs, failure_threshold=5, recovery_timeout=60):
        """每个模型配置配一个独立熔断器"""
        self._configs = sorted(model_configs, key=lambda c: c.priority)  # 按优先级排
        self._breakers = {cfg.model_id: CircuitBreaker(...) for cfg in model_configs}
        # ↑ 每个模型一个熔断器，互不影响

    def _select_candidates(self, model_preference):
        """
        候选排序规则：
        1. 如果有明确指定 model_preference → 优先用那个
        2. 按 priority 分组（priority=0 > priority=1）
        3. 同组内按 weight 加权随机（weight 越大越容易被选到）
        """
        # 同优先级内的加权随机算法：
        # score = random.random() ** (1.0 / weight)
        # 假设 weight=3:  0.7^(1/3) = 0.89 → 几乎总在最前
        # 假设 weight=0.5: 0.7^(1/0.5) = 0.49 → 大概率靠后
        scored = [(cfg, random.random() ** (1.0 / max(cfg.weight, 0.01))) for cfg in group]
        scored.sort(key=lambda x: -x[1])

    async def chat(self, messages, model_preference=None, **kwargs):
        """按候选顺序尝试，失败自动下一个"""
        candidates = self._select_candidates(model_preference)

        for cfg in candidates:
            breaker = self._breakers[cfg.model_id]  # 这个模型的熔断器
            try:
                return await breaker.call(self._try_model, cfg.model_id, messages, **kwargs)
            except RuntimeError:
                continue  # 熔断器拒绝 → 下一个
            except (APIError, RateLimitError, TimeoutError):
                continue  # API 异常 → 下一个

        raise RuntimeError("所有候选模型均不可用")
```
